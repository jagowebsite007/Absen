/** * ========================================== * Code.gs - REST API Backend & Google Drive Storage * ========================================== */ function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Payload POST kosong atau tidak valid." })).setMimeType(ContentService.MimeType.JSON); } var request = JSON.parse(e.postData.contents); var action = request.action; var payload = request.payload || {}; var result = { success: false, message: "Aksi tidak dikenal (" + action + ")" }; if (action === "login") { result = handleLogin(payload.nama, payload.pass); } else if (action === "submitAttendance") { result = handleSubmitAttendance(payload.userId, payload.tipe, payload.selfieData, payload.coords); } else if (action === "getAttendanceData") { result = handleGetAttendanceData(); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Terjadi kesalahan server: " + err.toString() })).setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { var params = e ? e.parameter : {}; var action = params.action; if (action === "login") { var res = handleLogin(params.nama, params.pass); return ContentService.createTextOutput(JSON.stringify(res)).setMimeType(ContentService.MimeType.JSON); } else if (action === "getAttendanceData") { var res = handleGetAttendanceData(); return ContentService.createTextOutput(JSON.stringify(res)).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput("REST API Absensi Rajawali Aktif!"); } catch (err) { return ContentService.createTextOutput("Error: " + err.message); } } function handleLogin(nama, pass) { try { if (!nama || !pass) { return { success: false, message: "Nama Pengguna dan Password harus diisi!" }; } var ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID); var sheet = ss.getSheetByName("User"); if (!sheet) { return { success: false, message: "Sheet 'User' belum ditemukan! Silakan jalankan tesOtorisasiDrive() / initialSetup() pada setup.gs." }; } var data = sheet.getDataRange().getValues(); if (data.length <= 1) { return { success: false, message: "Database pengguna kosong! Harap tambahkan baris pengguna pada Sheet 'User'." }; } var inputNama = String(nama).trim().toLowerCase(); var inputPass = String(pass).trim(); for (var i = 1; i < data.length; i++) { var sheetNama = String(data[i][0]).trim(); var sheetPass = String(data[i][1]).trim(); var sheetRole = String(data[i][2]).trim(); if (sheetNama.toLowerCase() === inputNama && sheetPass === inputPass) { var todayStatus = getTodayUserAttendance(sheetNama); var userSchoolConfig = getUserSchoolConfig(sheetNama); return { success: true, user: { id: sheetNama, nama: sheetNama, role: sheetRole || "Guru" }, todayStatus: todayStatus, schoolConfig: userSchoolConfig }; } } return { success: false, message: "Nama Pengguna atau Password salah! Periksa huruf kapital dan data di Sheet 'User'." }; } catch (err) { return { success: false, message: "Gagal verifikasi login: " + err.message }; } } /** * Membaca konfigurasi GPS Sekolah langsung dari Sheet User berdasarkan Nama User */ function getUserSchoolConfig(userId) { var config = { lat: CONFIG.SCHOOL_LAT || -2.53871, lng: CONFIG.SCHOOL_LNG || 140.70123, maxRadius: CONFIG.MAX_RADIUS_METERS || 20 }; try { var ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID); var sheet = ss.getSheetByName("User"); if (!sheet) return config; var data = sheet.getDataRange().getValues(); var targetUser = String(userId).trim().toLowerCase(); for (var i = 1; i < data.length; i++) { var sheetNama = String(data[i][0]).trim().toLowerCase(); if (sheetNama === targetUser) { var latVal = parseFloat(data[i][3]); var lngVal = parseFloat(data[i][4]); var radVal = parseFloat(data[i][5]); if (!isNaN(latVal) && latVal !== 0) config.lat = latVal; if (!isNaN(lngVal) && lngVal !== 0) config.lng = lngVal; if (!isNaN(radVal) && radVal > 0) config.maxRadius = radVal; break; } } } catch (e) { Logger.log("Gagal membaca koordinat custom user: " + e.message); } return config; } function getTodayUserAttendance(userId) { var status = { hasCheckedIn: false, hasCheckedOut: false }; try { var ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID); var sheet = ss.getSheetByName("Absensi"); if (!sheet) return status; var data = sheet.getDataRange().getValues(); var todayStr = Utilities.formatDate(new Date(), CONFIG.TIMEZONE, "yyyy-MM-dd"); for (var i = 1; i < data.length; i++) { var rawTime = data[i][0]; var rowUser = String(data[i][1]); var rowTipe = String(data[i][2]); var dateStr = ""; if (rawTime instanceof Date) { dateStr = Utilities.formatDate(rawTime, CONFIG.TIMEZONE, "yyyy-MM-dd"); } else { dateStr = String(rawTime).split(" ")[0]; } if (rowUser.toLowerCase() === String(userId).toLowerCase() && dateStr === todayStr) { if (rowTipe === "Check-In") status.hasCheckedIn = true; if (rowTipe === "Check-Out") status.hasCheckedOut = true; } } } catch(e) {} return status; } function handleSubmitAttendance(userId, tipe, selfieData, coords) { try { if (!selfieData) { return { success: false, message: "Gagal: Anda wajib mengambil foto selfie sebelum absen!" }; } var todayStatus = getTodayUserAttendance(userId); if (tipe === "Check-In" && todayStatus.hasCheckedIn) { return { success: false, message: "Gagal: Anda sudah melakukan Check-In untuk hari ini!" }; } if (tipe === "Check-Out" && todayStatus.hasCheckedOut) { return { success: false, message: "Gagal: Anda sudah melakukan Check-Out untuk hari ini!" }; } if (!coords || coords === "-") { return { success: false, message: "Gagal: Lokasi GPS tidak terdeteksi!" }; } var parts = coords.split(","); var userLat = parseFloat(parts[0]); var userLng = parseFloat(parts[1]); var userSchoolConfig = getUserSchoolConfig(userId); var schoolLat = userSchoolConfig.lat; var schoolLng = userSchoolConfig.lng; var maxRadius = userSchoolConfig.maxRadius; var distanceMeters = calculateHaversineDistance(userLat, userLng, schoolLat, schoolLng); if (distanceMeters > maxRadius) { return { success: false, message: "Gagal: Anda berada di luar area sekolah (" + Math.round(distanceMeters) + " meter). Radius maksimal " + maxRadius + " meter!" }; } var now = new Date(); var timeFormatted = Utilities.formatDate(now, CONFIG.TIMEZONE, "yyyy-MM-dd HH:mm:ss"); var dateFileFormatted = Utilities.formatDate(now, CONFIG.TIMEZONE, "yyyyMMdd_HHmmss"); var folder = DriveApp.getFolderById(CONFIG.FOLDER_ID); var cleanBase64 = selfieData.replace(/^data:image\/(png|jpeg|jpg);base64,/, ""); var imageBytes = Utilities.base64Decode(cleanBase64); var fileName = "ABSEN_" + tipe + "_" + userId + "_" + dateFileFormatted + ".jpg"; var blob = Utilities.newBlob(imageBytes, "image/jpeg", fileName); var driveFile = folder.createFile(blob); driveFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); var photoUrl = driveFile.getUrl(); var ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID); var sheet = ss.getSheetByName("Absensi"); if (!sheet) { initialSetup(); sheet = ss.getSheetByName("Absensi"); } sheet.appendRow([ timeFormatted, userId, tipe, coords + " (" + Math.round(distanceMeters) + "m)", photoUrl ]); var updatedStatus = getTodayUserAttendance(userId); return { success: true, message: "Absensi " + tipe + " Berhasil! Foto tersimpan di Google Drive.", photoUrl: photoUrl, todayStatus: updatedStatus }; } catch (err) { return { success: false, message: "Gagal menyimpan absensi: " + err.message }; } } function calculateHaversineDistance(lat1, lon1, lat2, lon2) { var R = 6371000; var dLat = (lat2 - lat1) * Math.PI / 180; var dLon = (lon2 - lon1) * Math.PI / 180; var a = Math.sin(dLat / 2) * Math.sin(dLat / 2) + Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.sin(dLon / 2) * Math.sin(dLon / 2); var c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a)); return R * c; } function handleGetAttendanceData() { try { var ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID); var sheet = ss.getSheetByName("Absensi"); if (!sheet) { return { success: true, data: [] }; } var data = sheet.getDataRange().getValues(); if (data.length <= 1) { return { success: true, data: [] }; } var list = []; for (var i = 1; i < data.length; i++) { var row = data[i]; var rawTime = row[0]; var formattedTime = rawTime; if (rawTime instanceof Date) { formattedTime = Utilities.formatDate(rawTime, CONFIG.TIMEZONE, "yyyy-MM-dd HH:mm:ss"); } else { formattedTime = String(rawTime); } list.push({ waktu: formattedTime, nama: String(row[1]), tipe: String(row[2]), gps: String(row[3]), fotoUrl: String(row[4]) }); } return { success: true, data: list }; } catch (err) { return { success: false, message: "Gagal mengambil data rekap: " + err.message }; } }
